template <typename T>
class DoublyLinkedList {
private:
    struct Node {
        T data;
        Node* prev{nullptr};
        Node* next{nullptr};
        Node(const T& val) : data(val) {}
    };
    Node* head{nullptr};
    Node* tail{nullptr};
    size_t list_size{0};

public:
    DoublyLinkedList() = default;

    void push_front(const T& value) {
        Node* newNode = new Node(value);
        if (!head) {
            head = tail = newNode;
        } else {
            newNode->next = head;
            head->prev = newNode;
            head = newNode;
        }
        ++list_size;
    }

    void reverseList() noexcept {
        if (!head || head == tail) return;
        Node* current = head;
        Node* temp = nullptr;
        while (current != nullptr) {
            temp = current->prev;
            current->prev = current->next;
            current->next = temp;
            current = current->prev;
        }
        if (temp != nullptr) {
            tail = head;
            head = temp->prev;
        }
    }

    ~DoublyLinkedList() {
        Node* current = head;
        while (current) {
            Node* nextNode = current->next;
            delete current;
            current = nextNode;
        }
        head = tail = nullptr;
        list_size = 0;
    }
};
